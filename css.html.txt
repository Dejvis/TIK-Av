        function updateProgress() {
            const progressPercent = Math.round((currentSlide / totalSlides) * 100);
            document.getElementById('progressPercent').textContent = `${progressPercent}%`;

        function showSlide(n) {
            const slides = document.querySelectorAll('.slide');
            slides.forEach(slide => slide.classList.remove('active'));
            document.getElementById(`slide${n}`).classList.add('active');
            currentSlide = n;
            updateProgress();

            // Scroll to top when changing slides
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function nextSlide() {
            if (currentSlide < totalSlides) {
                showSlide(currentSlide + 1);
            }
        }

        function prevSlide() {
            if (currentSlide > 1) {

        // Quiz functionality
        function checkAnswer(button, isCorrect) {
            const questionDiv = button.closest('[id^="quizQuestion"]');
            const feedbackId = questionDiv.id.replace("Question", "Feedback");
            const feedbackDiv = document.getElementById(feedbackId);

            // Reset all buttons in this question
            const allButtons = questionDiv.querySelectorAll('.question-option');
            allButtons.forEach(btn => {
                btn.classList.remove('correct', 'incorrect');
            });

            // Mark the clicked button
            if (isCorrect) {
                button.classList.add('correct');
                feedbackDiv.classList.remove('hidden');
            } else {
                button.classList.add('incorrect');

                // Find and show the correct answer
                const correctButton = [...allButtons].find(btn => btn.classList.contains('question-option') && btn !== button);
                if (correctButton) {
                    setTimeout(() => {
                        correctButton.classList.add('correct');
                        feedbackDiv.classList.remove('hidden');
                    }, 500);
                }
            }
        }

        // Flexbox demo functionality
        function updateFlexDemo(property, value) {
            const flexDemo = document.getElementById('flexDemo');
            
            switch(property) {
                case 'direction':
                    flexDemo.className = flexDemo.className.replace(/flex-row|flex-row-reverse|flex-col|flex-col-reverse/, 'flex-' + value);
                    break;
                case 'justify':
                    flexDemo.className = flexDemo.className.replace(/justify-start|justify-end|justify-center|justify-between|justify-around|justify-evenly/, 'justify-' + value);
                    break;
                case 'align':
                    flexDemo.className = flexDemo.className.replace(/items-start|items-end|items-center|items-baseline|items-stretch/, 'items-' + value);
                    break;
                case 'wrap':
                    flexDemo.className = flexDemo.className.replace(/flex-nowrap|flex-wrap|flex-wrap-reverse/, 'flex-' + value);
                    break;
            }
        }

        // Grid demo functionality
        function updateGridDemo(property, value) {
            const gridDemo = document.getElementById('gridDemo');
            
            switch(property) {
                case 'template':
                    if (value === '3cols') {
                        gridDemo.className = gridDemo.className.replace(/grid-cols-\d+/, 'grid-cols-3');
                    } else if (value === '2rows') {
                        gridDemo.className = gridDemo.className.replace(/grid-cols-\d+/, 'grid-cols-2');
                    }
                    break;
                case 'justify':
                    gridDemo.className = gridDemo.className.replace(/justify-start|justify-end|justify-center|justify-between|justify-around|justify-evenly/, 'justify-' + value);
                    break;
                case 'align':
                    gridDemo.className = gridDemo.className.replace(/items-start|items-end|items-center|items-baseline|items-stretch/, 'items-' + value);
                    break;
                case 'gap':
                    gridDemo.style.gap = value;
                    break;
            }
        }
