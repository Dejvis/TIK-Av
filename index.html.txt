        function loadModule() {
            const module = currentCourse.modules[currentModuleIndex];
            
            currentModuleTitle.textContent = module.title;
            currentModuleDescription.textContent = module.description;
            moduleContent.innerHTML = module.content;
            
            // Update navigation buttons
            prevModuleBtn.disabled = currentModuleIndex === 0;
            nextModuleBtn.disabled = currentModuleIndex === currentCourse.modules.length - 1;
            
            // Update progress
            const progress = ((currentModuleIndex + 1) / currentCourse.modules.length) * 100;
            progressBar.style.width = `${progress}%`;
            progressPercent.textContent = `${Math.round(progress)}%`;
            
            // Update completed modules
            document.querySelectorAll('#course-modules div').forEach((item, index) => {
                const checkIcon = item.querySelector('.fa-check-circle');
                if (index <= currentModuleIndex) {
                    checkIcon.classList.remove('hidden');
                } else {
                    checkIcon.classList.add('hidden');
                }
            });
        }
