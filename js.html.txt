for (let i = 0; i < fruits.length; i++) {

}

// For...of loop (ES6)
for (const fruit of fruits) {
    console.log(fruit);
}</code></pre>

while (count < 5) {
    console.log(count);
    count++;
}

// Do...while loop
let x = 0;
do {
    console.log(x);
    x++;
} while (x < 5);</code></pre>

                <h2 class="text-2xl font-bold mb-4 text-js-yellow">Functions</h2>
                <div class="mb-6">
                    <p class="mb-4">Functions are reusable blocks of code that perform a specific task.</p>
                    
                    <div class="bg-gray-100 dark:bg-gray-700 p-4 rounded-lg mb-4 overflow-x-auto">
                        <pre class="code-block text-sm"><code>// Function declaration
function greet(name) {
    return `Hello, ${name}!`;
}

// Function expression
const greet = function(name) {
    return `Hello, ${name}!`;
};

// Arrow function (ES6)
const greet = (name) => `Hello, ${name}!`;

// Calling functions
console.log(greet("Alice")); // Output: Hello, Alice!</code></pre>
                    </div>
                    
                    <h3 class="text-xl font-semibold mt-4 mb-2">Default Parameters</h3>
                    <div class="bg-gray-100 dark:bg-gray-700 p-4 rounded-lg mb-4 overflow-x-auto">
                        <pre class="code-block text-sm"><code>function greet(name = "World") {
    return `Hello, ${name}!`;
}

console.log(greet());      // Output: Hello, World!
console.log(greet("Bob")); // Output: Hello, Bob!</code></pre>
                    </div>
                    
                    <h3 class="text-xl font-semibold mt-4 mb-2">Higher-Order Functions</h3>
                    <div class="bg-gray-100 dark:bg-gray-700 p-4 rounded-lg mb-4 overflow-x-auto">
                        <pre class="code-block text-sm"><code>// Function that takes another function as argument
function applyOperation(x, y, operation) {
    return operation(x, y);
}

const sum = (a, b) => a + b;
const multiply = (a, b) => a * b;

console.log(applyOperation(5, 3, sum));      // 8
console.log(applyOperation(5, 3, multiply)); // 15</code></pre>
                    </div>

            document.getElementById('progressText').textContent = `Lesson ${currentSlide} of ${totalSlides}`;
        }

        function showSlide(n) {
            const slides = document.querySelectorAll('.slide');
            slides.forEach(slide => slide.classList.remove('active'));
            document.getElementById(`slide${n}`).classList.add('active');
            currentSlide = n;
            updateProgress();
            
            // Scroll to top when changing slides
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function nextSlide() {
            if (currentSlide < totalSlides) {
                showSlide(currentSlide + 1);
            }
        }

        function prevSlide() {
            if (currentSlide > 1) {
                showSlide(currentSlide - 1);
            }
        }

        // Quiz functionality
        function checkAnswer(button, isCorrect) {
            const questionDiv = button.closest('[id^="quizQuestion"]');
            const feedbackId = questionDiv.id.replace("Question", "Feedback");
            const feedbackDiv = document.getElementById(feedbackId);
            
            // Reset all buttons in this question
            const allButtons = questionDiv.querySelectorAll('.question-option');
            allButtons.forEach(btn => {
                btn.classList.remove('correct', 'incorrect');
            });
            
            // Mark the clicked button
            if (isCorrect) {
                button.classList.add('correct');
                feedbackDiv.classList.remove('hidden');
            } else {
                button.classList.add('incorrect');
                
                // Find and show the correct answer
                const correctButton = [...allButtons].find(btn => btn.classList.contains('question-option') && btn !== button);
                if (correctButton) {
                    setTimeout(() => {
                        correctButton.classList.add('correct');
                        feedbackDiv.classList.remove('hidden');
                    }, 500);
                }
            }
        }

        // Playground functionality
        function runCode() {
            const code = document.getElementById('playgroundCode').value;
            const outputDiv = document.getElementById('playgroundOutput');
            
            // Clear previous output
            outputDiv.innerHTML = '';
            
            try {
                // Create a function from the code and execute it
                const func = new Function(code);
                
                // Override console.log to capture output
                const oldLog = console.log;
                console.log = function(...args) {
                    oldLog.apply(console, args); // Keep original logging
                    args.forEach(arg => {
                        const line = document.createElement('div');
                        line.textContent = String(arg);
                        outputDiv.appendChild(line);
                    });
                };
                
                // Execute the code
                func();
                
                // Restore original console.log
                console.log = oldLog;
                
            } catch (error) {
                const errorDiv = document.createElement('div');
                errorDiv.className = 'text-red-500';
                errorDiv.textContent = error.message;
                outputDiv.appendChild(errorDiv);
            }
        }
